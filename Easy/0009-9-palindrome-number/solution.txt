class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        int copyX = x;
        int reversed = 0;

        while (copyX > 0 ){
            int remainder = copyX % 10;
            int popEnd = copyX/10;
            reversed = (reversed * 10) + remainder;
            copyX = copyX/10;
        }

        return x == reversed;
        
    }
}
